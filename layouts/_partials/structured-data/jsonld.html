{{- /* Renders one JSON-LD script with a schema.org @graph for the current page: Organization, WebSite, WebPage (or CollectionPage), BreadcrumbList, and, depending on the page, Article, Product and FAQPage. Every node is built as a dict and serialized with jsonify, so the output is always valid JSON. Call it from the head of your base template: {{ partial "structured-data/jsonld.html" . }} Front matter of a page: structuredData: false # no JSON-LD on this page structuredData: type: Product # or Article, WebPage; default depends on the section product: brand: Acme sku: AC-100 mpn: 100-200 category: Insulation testers offers: # optional price: 1250 priceCurrency: EUR availability: InStock faq: # adds a FAQPage - question: How long is the warranty? answer: Two years. @context {page} . */ -}} {{- if ne .Params.structuredData false -}} {{- $cfg := site.Params.structuredData | default dict -}} {{- $pcfg := .Params.structuredData | default dict -}} {{- $locale := .Language.Locale -}} {{- $graph := slice -}} {{- $images := partial "structured-data/images.html" . -}} {{- $desc := .Description | default (string .Summary) | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- /* Organization: one @id for every language, on the default site's home. */ -}} {{- $orgID := "" -}} {{- $org := $cfg.organization | default dict -}} {{- with $org.name -}} {{- $root := (index hugo.Sites 0).Home.Permalink -}} {{- $orgID = printf "%s#organization" $root -}} {{- $o := dict "@type" ($org.type | default "Organization") "@id" $orgID "name" . "url" ($org.url | default $root) -}} {{- range $k := slice "legalName" "email" "telephone" "vatID" -}} {{- with index $org $k -}}{{- $o = merge $o (dict $k .) -}}{{- end -}} {{- end -}} {{- with $org.logo -}} {{- $o = merge $o (dict "logo" (dict "@type" "ImageObject" "url" (. | absURL))) -}} {{- end -}} {{- with $org.sameAs -}}{{- $o = merge $o (dict "sameAs" .) -}}{{- end -}} {{- with $org.address -}} {{- $a := dict "@type" "PostalAddress" -}} {{- range $k, $v := . -}} {{- with $v -}}{{- $a = merge $a (dict $k .) -}}{{- end -}} {{- end -}} {{- if gt (len $a) 1 -}}{{- $o = merge $o (dict "address" $a) -}}{{- end -}} {{- end -}} {{- $graph = $graph | append $o -}} {{- end -}} {{- /* WebSite for the current language. */ -}} {{- $siteID := printf "%s#website" site.Home.Permalink -}} {{- $ws := dict "@type" "WebSite" "@id" $siteID "url" site.Home.Permalink "name" site.Title "inLanguage" $locale -}} {{- with $orgID -}}{{- $ws = merge $ws (dict "publisher" (dict "@id" .)) -}}{{- end -}} {{- $graph = $graph | append $ws -}} {{- /* WebPage */ -}} {{- $pageID := printf "%s#webpage" .Permalink -}} {{- $wp := dict "@type" (cond .IsPage "WebPage" "CollectionPage") "@id" $pageID "url" .Permalink "name" .Title "inLanguage" $locale "isPartOf" (dict "@id" $siteID) -}} {{- with $desc -}}{{- $wp = merge $wp (dict "description" .) -}}{{- end -}} {{- if not .Date.IsZero -}}{{- $wp = merge $wp (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00")) -}}{{- end -}} {{- if not .Lastmod.IsZero -}}{{- $wp = merge $wp (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) -}}{{- end -}} {{- with $images -}}{{- $wp = merge $wp (dict "primaryImageOfPage" (dict "@type" "ImageObject" "url" (index . 0))) -}}{{- end -}} {{- /* BreadcrumbList */ -}} {{- if and (ne $cfg.breadcrumbs false) (not .IsHome) -}} {{- $items := slice -}} {{- $pos := 0 -}} {{- range .Ancestors.Reverse -}} {{- $pos = add $pos 1 -}} {{- $items = $items | append (dict "@type" "ListItem" "position" $pos "name" .LinkTitle "item" .Permalink) -}} {{- end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $pos 1) "name" .LinkTitle) -}} {{- $bcID := printf "%s#breadcrumb" .Permalink -}} {{- $graph = $graph | append (dict "@type" "BreadcrumbList" "@id" $bcID "itemListElement" $items) -}} {{- $wp = merge $wp (dict "breadcrumb" (dict "@id" $bcID)) -}} {{- end -}} {{- /* Main entity: Article for pages in the article sections, Product on request. */ -}} {{- $type := $pcfg.type -}} {{- if not $type -}} {{- $sections := $cfg.articleSections | default site.MainSections -}} {{- if and .IsPage (in $sections .Section) -}} {{- $type = $cfg.articleType | default "Article" -}} {{- end -}} {{- end -}} {{- if in (slice "Article" "BlogPosting" "NewsArticle" "TechArticle") $type -}} {{- $a := dict "@type" $type "@id" (printf "%s#article" .Permalink) "headline" (.Title | truncate 110 "") "mainEntityOfPage" (dict "@id" $pageID) "inLanguage" $locale -}} {{- with $desc -}}{{- $a = merge $a (dict "description" .) -}}{{- end -}} {{- with $images -}}{{- $a = merge $a (dict "image" .) -}}{{- end -}} {{- if not .Date.IsZero -}}{{- $a = merge $a (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00")) -}}{{- end -}} {{- if not .Lastmod.IsZero -}}{{- $a = merge $a (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) -}}{{- end -}} {{- $authors := slice -}} {{- with .Params.author -}} {{- range (cond (reflect.IsSlice .) . {{- $a = merge $a (dict "author" (dict "@id" $orgID)) -}} {{- end -}} {{- with $orgID -}}{{- $a = merge $a (dict "publisher" (dict "@id" .)) -}}{{- end -}} {{- $graph = $graph | append $a -}} {{- else if eq $type "Product" -}} {{- $p := $pcfg.product | default dict -}} {{- $pr := dict "@type" "Product" "@id" (printf "%s#product" .Permalink) "name" .Title "url" .Permalink -}} {{- with $desc -}}{{- $pr = merge $pr (dict "description" .) -}}{{- end -}} {{- with $images -}}{{- $pr = merge $pr (dict "image" .) -}}{{- end -}} {{- range $k := slice "sku" "mpn" "gtin" "category" "model" -}} {{- with $p.manufacturer -}} {{- $pr = merge $pr (dict "manufacturer" (dict "@type" "Organization" "name" .)) -}} {{- end -}} {{- with $p.offers -}} {{- $of := dict "@type" "Offer" "url" $.Permalink -}} {{- merge $of (dict "availability" (printf "https://schema.org/%s" .)) -}}{{- end -}} {{- with $orgID -}}{{- $of = merge $of (dict "seller" (dict "@id" .)) -}}{{- end -}} {{- $pr = merge $pr (dict "offers" $of) -}} {{- end -}} {{- $graph = $graph | append $pr -}} {{- $wp = merge $wp (dict "mainEntity" (dict "@id" (printf "%s#product" .Permalink))) -}} {{- end -}} {{- $graph = $graph | append "FAQPage" "@id" (printf "%s#faq" $.Permalink) "mainEntity" . "inLanguage" $locale "isPartOf" (dict "@id" $pageID)) -}} {{- end -}} {{- end -}} {{- end -}}
